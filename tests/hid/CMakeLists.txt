set(PB_SRC "${PROJECT_SOURCE_DIR}/src")
file(GLOB PB_HID_TEST_SOURCES CONFIGURE_DEPENDS "${PB_SRC}/hid/hid_*.c")
add_executable(test_hid test_hid.c ${PB_HID_TEST_SOURCES} "${PB_SRC}/hid/replay.c" "${PB_SRC}/record/record.c"
    "${PB_SRC}/core/out.c" "${PB_SRC}/core/sys.c" "${PB_SRC}/core/pb.c")
target_include_directories(test_hid PRIVATE "${PB_SRC}" "${PB_SRC}/hid")
target_link_libraries(test_hid PRIVATE hid-parser::hid-parser wirefile::wirefile vpad::vpad Threads::Threads)
if(UNIX)
    target_compile_definitions(test_hid PRIVATE _POSIX_C_SOURCE=200809L)
endif()
if(CMAKE_C_COMPILER_ID MATCHES "GNU|Clang")
    target_compile_options(test_hid PRIVATE -Wall -Wextra -Wpedantic)
endif()

add_test(NAME hid.replay COMMAND test_hid replay ${CMAKE_CURRENT_BINARY_DIR})
add_test(NAME hid.uhid COMMAND test_hid uhid)
set_tests_properties(hid.uhid PROPERTIES SKIP_RETURN_CODE 77)
add_test(NAME hid.list COMMAND padbridge list)
add_test(NAME hid.record_usage COMMAND padbridge record x)
set_tests_properties(hid.record_usage PROPERTIES WILL_FAIL TRUE)
add_test(NAME hid.record_missing COMMAND padbridge record hid:/dev/nonexistent ${CMAKE_CURRENT_BINARY_DIR}/none.pcapng)
set_tests_properties(hid.record_missing PROPERTIES PASS_REGULAR_EXPRESSION "cannot open")
