set(PB_TMP "${CMAKE_CURRENT_BINARY_DIR}/home")
file(MAKE_DIRECTORY "${PB_TMP}/run" "${PB_TMP}/config")
set(PB_ENV "XDG_RUNTIME_DIR=${PB_TMP}/run;XDG_CONFIG_HOME=${PB_TMP}/config;HOME=${PB_TMP}")

add_test(NAME core.status_idle COMMAND padbridge status)
set_tests_properties(core.status_idle PROPERTIES ENVIRONMENT "${PB_ENV}" WILL_FAIL TRUE)
add_test(NAME core.stop_idle COMMAND padbridge stop)
set_tests_properties(core.stop_idle PROPERTIES ENVIRONMENT "${PB_ENV}" WILL_FAIL TRUE)
add_test(NAME core.unknown_command COMMAND padbridge nope)
set_tests_properties(core.unknown_command PROPERTIES WILL_FAIL TRUE)
add_test(NAME core.bad_config COMMAND padbridge run --config ${CMAKE_CURRENT_SOURCE_DIR}/bad.conf x)
set_tests_properties(core.bad_config PROPERTIES PASS_REGULAR_EXPRESSION "bad setting")
add_test(NAME core.version COMMAND padbridge --version)
set_tests_properties(core.version PROPERTIES PASS_REGULAR_EXPRESSION "padbridge [0-9]")
